schema AddressDto
  properties: streetAddress (string), postalCode (string), city (string)
  required: city, postalCode, streetAddress

schema CustomerInfoDto
  properties: customerId (string), firstname (string), lastname (string), contactAddress (AddressDto), billingAddress (AddressDto)
  required: billingAddress, contactAddress, customerId, firstname, lastname

schema InsuranceOptionsDto
  properties: startDate (string), insuranceType (string), deductible (MoneyAmountDto)
  required: deductible, insuranceType, startDate

schema InsuranceQuoteDto
  properties: expirationDate (string), insurancePremium (MoneyAmountDto), policyLimit (MoneyAmountDto)
  required: expirationDate, insurancePremium, policyLimit

schema InsuranceQuoteRequestDto
  properties: id (integer), date (string), statusHistory (array of RequestStatusChangeDto), customerInfo (CustomerInfoDto), insuranceOptions (InsuranceOptionsDto), insuranceQuote (InsuranceQuoteDto), policyId (string)
  required: customerInfo, insuranceOptions

schema MoneyAmountDto
  properties: amount (number), currency (string)
  required: amount, currency

schema RequestStatusChangeDto
  properties: date (string), status (string)
  required: status

  required: birthday, city, firstname, lastname, postalCode, streetAddress
