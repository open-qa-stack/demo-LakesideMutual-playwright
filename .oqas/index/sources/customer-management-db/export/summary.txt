table public.interactionlogs
  columns: customer_id character varying(255) not null, last_acknowledged_interaction_id character varying(255), username character varying(255)
  primary key: customer_id

table public.interactionlogs_interactions
  columns: interaction_log_aggregate_root_customer_id character varying(255) not null, interactions_id character varying(255) not null
  foreign keys: interactions_id -> public.interactions (id); interaction_log_aggregate_root_customer_id -> public.interactionlogs (customer_id)
  constraints: UNIQUE (interactions_id)

table public.interactions
  columns: sent_by_operator boolean not null, date timestamp(6) without time zone, content character varying(255), id character varying(255) not null
  primary key: id
